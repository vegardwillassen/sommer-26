function fpUrl(name){ return "https://commons.wikimedia.org/wiki/Special:FilePath/"+encodeURIComponent(name)+"?width=640"; }
function parseCommons(j){
  const pages = (j && j.query && j.query.pages) ? Object.values(j.query.pages) : [];
  pages.sort((a,b)=>(a.index||99)-(b.index||99));
  return pages.map(p=>p.imageinfo && p.imageinfo[0]).filter(Boolean)
    .filter(ii=>/image\/(jpeg|png)/.test(ii.mime))
    .map(ii=>ii.thumburl || ii.url);
}
function commonsApiUrl(q2, jsonpCb){
  return "https://commons.wikimedia.org/w/api.php?action=query&format=json"+
    (jsonpCb ? "&callback="+jsonpCb : "&origin=*")+
    "&generator=search&gsrnamespace=6&gsrlimit=12&gsrsearch="+encodeURIComponent(q2)+
    "&prop=imageinfo&iiprop=url%7Cmime&iiurlwidth=640";
}
async function fetchCommons(q2){
  const r = await fetch(commonsApiUrl(q2));
  return parseCommons(await r.json());
}
let jsonpN = 0;
function jsonpCommons(q2){
  return new Promise((res, rej)=>{
    const cb = "napaCb"+(jsonpN++);
    const s = document.createElement("script");
    const t = setTimeout(()=>{
      window[cb] = function(){ try{ delete window[cb]; }catch(e){} };
      s.remove(); rej(new Error("timeout"));
    }, 8000);
    window[cb] = j=>{ clearTimeout(t); try{ delete window[cb]; }catch(e){} s.remove(); res(parseCommons(j)); };
    s.onerror = ()=>{ clearTimeout(t); try{ delete window[cb]; }catch(e){} s.remove(); rej(new Error("jsonp failed")); };
    s.src = commonsApiUrl(q2, cb);
    document.head.appendChild(s);
  });
}
function probeImages(names){
  return new Promise(res=>{
    const out = new Array(names.length);
    let pending = names.length;
    if(!pending) return res([]);
    names.forEach((n,i)=>{
      const url = fpUrl(n);
      const im = new Image();
      let called = false;
      const done = ok=>{
        if(called) return;
        called = true;
        im.onload = im.onerror = null;
        if(ok) out[i] = url;
        if(--pending===0) res(out.filter(Boolean));
      };
      const t = setTimeout(()=>done(false), 7000);
      im.onload = ()=>{ clearTimeout(t); done(true); };
      im.onerror = ()=>{ clearTimeout(t); done(false); };
      im.src = url;
    });
  });
}

async function getGalleryUrls(it){
  if(GALLERY[it.id] && GALLERY[it.id].length) return GALLERY[it.id];
  if(GAL_CACHE[it.id] && GAL_CACHE[it.id].length) return GAL_CACHE[it.id];
  if(GAL_PENDING[it.id]) return GAL_PENDING[it.id];
  const job = (async ()=>{
    let urls = [];
    const qs = QUERIES[it.id] || [it.name + " Cyprus"];
    try{
      for(const q2 of qs){
        if(urls.length >= 5) break;
        const got = await fetchCommons(q2);
        for(const u of got){ if(urls.length < 5 && !urls.includes(u)) urls.push(u); }
      }
    }catch(e){}
    if(!urls.length){
      try{
        for(const q2 of qs){
          if(urls.length >= 5) break;
          const got = await jsonpCommons(q2);
          for(const u of got){ if(urls.length < 5 && !urls.includes(u)) urls.push(u); }
        }
      }catch(e){}
    }
    if(!urls.length && CANDIDATES[it.id]){
      try{ urls = (await probeImages(CANDIDATES[it.id])).slice(0,5); }catch(e){}
    }
    if(!urls.length && it.img && !it.imgFailed) urls = [it.img];
    return urls;
  })();
  GAL_PENDING[it.id] = job;
  const urls = await job;
  delete GAL_PENDING[it.id];
  if(urls.length){
    GAL_CACHE[it.id] = urls;
    lsSet("napafoodGal", JSON.stringify(GAL_CACHE));
  }
  return urls;
}

function renderGallery(it, urls){
  const mm = document.querySelector("#galWrap .media");
  const tw = document.getElementById("thumbs");
  if(!mm || !tw) return;
  let idx = 0;
  function show(i){
    idx = (i + urls.length) % urls.length;
    setMain(urls[idx]);
    const c = mm.querySelector(".gcount");
    if(c) c.textContent = (idx+1) + " / " + urls.length;
    tw.querySelectorAll("img").forEach((x,k)=>x.classList.toggle("sel", k===idx));
  }
  tw.innerHTML = urls.map((u,i)=>'<img src="'+esc(u)+'" class="'+(i===0?"sel":"")+'" alt="" onerror="this.style.display=\'none\'">').join("");
  tw.querySelectorAll("img").forEach((im,k)=>im.addEventListener("click", ev=>{ ev.stopPropagation(); show(k); }));
  if(urls.length > 1){
    mm.classList.add("has-nav");
    if(!mm.querySelector(".gnav")){
      const prev = document.createElement("button"); prev.className="gnav prev"; prev.innerHTML="&#8249;"; prev.setAttribute("aria-label","Previous photo");
      const next = document.createElement("button"); next.className="gnav next"; next.innerHTML="&#8250;"; next.setAttribute("aria-label","Next photo");
      const cnt = document.createElement("div"); cnt.className="gcount";
      prev.addEventListener("click", ev=>{ ev.stopPropagation(); show(idx-1); });
      next.addEventListener("click", ev=>{ ev.stopPropagation(); show(idx+1); });
      mm.appendChild(prev); mm.appendChild(next); mm.appendChild(cnt);
    }
    if(!mm.dataset.touchBound){
      mm.dataset.touchBound = "1";
      let tx = null;
      mm.addEventListener("touchstart", e=>{ tx = e.changedTouches[0].clientX; }, {passive:true});
      mm.addEventListener("touchend", e=>{
        if(tx===null) return;
        const dx = e.changedTouches[0].clientX - tx; tx = null;
        if(Math.abs(dx) > 40) show(idx + (dx<0 ? 1 : -1));
      }, {passive:true});
    }
  }
  show(0);
}

  if(!document.getElementById("galNote")) return;

  const m = document.getElementById("modal");

  const note = document.getElementById("galNote");
  const tw = document.getElementById("thumbs");
  if(!note || !tw) return;

  if(!it.img || it.imgFailed){ it.img = urls[0]; it.imgFailed = false; updateCardMedia(it); }
  renderGallery(it, urls);

    " of this place. Use the arrows or swipe.";
}

async function prefetchCovers(){
  for(let pass = 0; pass < 2; pass++){
    let i = 0;
    const items = DATA.filter(it=>!it.img || it.imgFailed);
    if(!items.length) break;
    await Promise.all(Array.from({length:2}, async ()=>{
      while(i < items.length){
        const it = items[i++];
        try{
          const urls = await getGalleryUrls(it);
          if(urls.length && (!it.img || it.imgFailed)){
            it.img = urls[0];
            it.imgFailed = false;
            updateCardMedia(it);
          }
        }catch(e){}
      }
    }));
    if(pass===0) await new Promise(r=>setTimeout(r, 4000));
  }
}

function toggle(id, act){
  if(act==="done"){
    if(state[id]==="done"){
      state[id] = wantOf(id) ? "want" : "";
      extra.unDone[id] = 1; /* remember the correction so it syncs to the partner */
    } else {
      state[id] = "done";
      delete extra.unDone[id];
      extra.notes[id] = extra.notes[id]||{};
      const n = extra.notes[id];
      /* refresh the date on redo ONLY when there is no real memory attached —
         a note or a spent amount means the original date is part of the record */
      if(!n.dt || !(n.t || extra.spent[id]!=null)) n.dt = dayLabelToday();
    }
  } else {
    const p = extra.who || "X";
    let s = wantOf(id);
    s = s.includes(p) ? s.replace(p,"") : s + p;
    if(s) extra.wantBy[id] = s; else delete extra.wantBy[id];
    if(state[id]!=="done") state[id] = s ? "want" : "";
  }
  saveState(); saveExtra();
  /* keep the grid steady under the user's finger: update the tapped card in
     place and re-sort only on the next full render (filter change etc.) */
  if(stat==="all"){ refreshCard(id); updateHeaderCounts(); }
  else render();
}

function chk36(s){
  let h = 0;
  for(let i=0;i<s.length;i++){ h = (h*31 + s.charCodeAt(i))>>>0; }
  return ("000000"+h.toString(36)).slice(-6);
}
function parse36(str){
  let n = 0n;
  for(const ch of str){
    const v = parseInt(ch, 36);
    if(isNaN(v)) throw new Error("bad char");
    n = n*36n + BigInt(v);
  }
  return n;
}
function encodeState(){
  if(extra.who==="V" || extra.who==="H"){
    let d = 0n, v = 0n, hh = 0n, dy = 0n, u = 0n;
    DATA.forEach((it,i)=>{
      const bit = 1n << BigInt(i);
      if(state[it.id]==="done") d |= bit;
      if(extra.unDone[it.id]) u |= bit; /* corrections: "this was NOT done after all" */
      const s = wantOf(it.id);
      if(s.includes("V") || (extra.who==="V" && s.includes("X"))) v |= bit;
      if(s.includes("H") || (extra.who==="H" && s.includes("X"))) hh |= bit;
      dy |= (BigInt(extra.days[it.id]||0) << BigInt(4*i));
    });
    const payload = extra.who+"."+DATA.length.toString(36)+"."+d.toString(36)+"."+v.toString(36)+"."+hh.toString(36)+"."+dy.toString(36)+"."+u.toString(36);
    return "FOOD5." + payload + "." + chk36(payload);
  }
  let w = 0n, d = 0n, dy = 0n;
  DATA.forEach((it,i)=>{
    const st = state[it.id];
    if(st==="want") w |= (1n << BigInt(i));
    if(st==="done") d |= (1n << BigInt(i));
    const day = extra.days[it.id]||0;
    dy |= (BigInt(day) << BigInt(4*i));
  });
  return "FOOD3." + w.toString(36) + "." + d.toString(36) + "." + dy.toString(36);
}
function decodeState(code){
  try{
    code = (code||"").trim();
    /* be forgiving: extract the code from a pasted WhatsApp/SMS message */
    const mx = code.match(/FOOD\d\.[0-9a-zA-Z.]+/);
    if(mx) code = mx[0].replace(/\.+$/,"");
    const incoming = {};
    const incomingDays = {};
    if(code.indexOf("FOOD5.")===0){
      const parts = code.slice(6).split(".");
      if(parts.length!==8) return {ok:false};
      const who = parts[0];
      const payload = parts.slice(0,7).join(".");
      if(chk36(payload)!==parts[7]) return {ok:false, bad:"That code has a typo in it, ask for a fresh one."};
      if(parseInt(parts[1],36)!==DATA.length) return {ok:false, bad:"The other phone has a different version of the list, so nothing was merged."};
      if(who!=="V" && who!=="H") return {ok:false};
      const d = parse36(parts[2]), v = parse36(parts[3]), hh = parse36(parts[4]), dy = parse36(parts[5]), u = parse36(parts[6]);
      let added = 0; const conflicts = [];
      DATA.forEach((it,i)=>{
        const bit = 1n << BigInt(i);
        if((d & bit) && state[it.id]!=="done"){ state[it.id] = "done"; delete extra.unDone[it.id]; added++; }
        /* the sender corrected an accidental "done" — honour it here too */
        if((u & bit) && state[it.id]==="done"){
          state[it.id] = wantOf(it.id) ? "want" : "";
          extra.unDone[it.id] = 1;
          added++;
        }
        const senderWants = who==="V" ? !!(v & bit) : !!(hh & bit);
        let s = wantOf(it.id);
        const had = s.includes(who);
        if(senderWants && !had){ s += who; added++; }
        if(!senderWants && had){ s = s.replace(who,""); added++; }
        if(s) extra.wantBy[it.id] = s; else delete extra.wantBy[it.id];
        if(state[it.id]!=="done") state[it.id] = s ? "want" : "";
        const day = Number((dy >> BigInt(4*i)) & 15n);
        if(day>=1 && day<=TRIP_LEN){
          if(!extra.days[it.id]){ extra.days[it.id] = day; added++; }
          else if(extra.days[it.id]!==day) conflicts.push(it.name+": "+DAY_LABELS[extra.days[it.id]-1]+" on this phone, "+DAY_LABELS[day-1]+" in the code");
        }
      });
      saveState(); saveExtra();
      return {ok:true, added, conflicts};
    }
    if(code.indexOf("FOOD4.")===0){
      const parts = code.slice(6).split(".");
      if(parts.length!==7) return {ok:false};
      const who = parts[0];
      const payload = parts.slice(0,6).join(".");
      if(chk36(payload)!==parts[6]) return {ok:false, bad:"That code has a typo in it, ask for a fresh one."};
      if(parseInt(parts[1],36)!==DATA.length) return {ok:false, bad:"The other phone has a different version of the list, so nothing was merged."};
      if(who!=="V" && who!=="H") return {ok:false};
      const d = parse36(parts[2]), v = parse36(parts[3]), hh = parse36(parts[4]), dy = parse36(parts[5]);
      let added = 0; const conflicts = [];
      DATA.forEach((it,i)=>{
        const bit = 1n << BigInt(i);
        if((d & bit) && state[it.id]!=="done"){ state[it.id] = "done"; added++; }
        const senderWants = who==="V" ? !!(v & bit) : !!(hh & bit);
        let s = wantOf(it.id);
        const had = s.includes(who);
        if(senderWants && !had){ s += who; added++; }
        if(!senderWants && had){ s = s.replace(who,""); added++; }
        if(s) extra.wantBy[it.id] = s; else delete extra.wantBy[it.id];
        if(state[it.id]!=="done") state[it.id] = s ? "want" : "";
        const day = Number((dy >> BigInt(4*i)) & 15n);
        if(day>=1 && day<=TRIP_LEN){
          if(!extra.days[it.id]){ extra.days[it.id] = day; added++; }
          else if(extra.days[it.id]!==day) conflicts.push(it.name+": "+DAY_LABELS[extra.days[it.id]-1]+" on this phone, "+DAY_LABELS[day-1]+" in the code");
        }
      });
      saveState(); saveExtra();
      return {ok:true, added, conflicts};
    }
    if(code.indexOf("FOOD3.")===0){
      const parts = code.slice(6).split(".");
      if(parts.length!==3) return {ok:false};
      const w = parse36(parts[0]), d = parse36(parts[1]), dy = parse36(parts[2]);
      DATA.forEach((it,i)=>{
        const bit = 1n << BigInt(i);
        if(d & bit) incoming[it.id] = "done";
        else if(w & bit) incoming[it.id] = "want";
        const day = Number((dy >> BigInt(4*i)) & 15n);
        if(day>=1 && day<=TRIP_LEN) incomingDays[it.id] = day;
      });
    } else if(code.indexOf("FOOD2.")===0){
      const parts = code.slice(6).split(".");
      if(parts.length!==2) return {ok:false};
      const w = parse36(parts[0]), d = parse36(parts[1]);
      DATA.forEach((it,i)=>{
        const bit = 1n << BigInt(i);
        if(d & bit) incoming[it.id] = "done";
        else if(w & bit) incoming[it.id] = "want";
      });
    } else if(code.indexOf("FOOD1.")===0){
      const o = JSON.parse(atob(code.slice(6)));
      (o.w||[]).forEach(id=>incoming[id]="want");
      (o.d||[]).forEach(id=>incoming[id]="done");
    } else return {ok:false};
    let added = 0;
    for(const id in incoming){
      const cur = state[id]||"";
      if(incoming[id]==="done" && cur!=="done"){ state[id]="done"; added++; }
      else if(incoming[id]==="want" && cur===""){ state[id]="want"; added++; }
      /* keep legacy wants inside the per-person model so they survive future exports */
      if(incoming[id]==="want" && !wantOf(id)) extra.wantBy[id] = "X";
    }
    for(const id in incomingDays){
      if(!extra.days[id]){ extra.days[id] = incomingDays[id]; added++; }
    }
    saveState(); saveExtra();
    return {ok:true, added};
  }catch(e){ return {ok:false}; }
}

  btn.textContent = text;

function doCopy(code, btn, revert){
  /* the paste-in field is for the PARTNER's code — only borrow it for the
     manual fallback, never leave our own code sitting there */
  const inp = document.getElementById("syncIn");
  const ok = ()=>flashBtn(btn, "Copied", revert);
  const manual = ()=>{
    inp.value = code; inp.focus(); inp.select();
    try{ document.execCommand("copy") ? ok() : flashBtn(btn, "Select & copy", revert); }catch(e){ flashBtn(btn, "Select & copy", revert); }
  };
  if(navigator.clipboard && navigator.clipboard.writeText){
    navigator.clipboard.writeText(code).then(ok).catch(manual);
  } else manual();
}
